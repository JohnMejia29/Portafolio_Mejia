				document.documentElement.lang = language;

				document.querySelectorAll("[data-i18n]").forEach((element) => {
					const key = element.getAttribute("data-i18n");
					if (key && dictionary[key] !== undefined) {
						element.textContent = dictionary[key];
					}
				});

				document.querySelectorAll("[data-i18n-attrs]").forEach((element) => {
					Array.from(element.attributes).forEach((attribute) => {
						const prefix = "data-i18n-attr-";
						if (!attribute.name.startsWith(prefix)) return;

						const translatedValue = dictionary[attribute.value];
						if (translatedValue === undefined) return;

						const targetAttribute = attribute.name.slice(prefix.length);
						element.setAttribute(targetAttribute, translatedValue);
					});
				});

				updateLanguageControl(language, dictionary);

				if (persist) {
					try {
						localStorage.setItem(languageStorageKey, language);
					} catch {
						// El cambio se conserva durante la sesión aunque no se pueda persistir.
					}
				}

				window.dispatchEvent(
					new CustomEvent("portfolio:translations-applied", {
						detail: { language },
					}),
				);
			};

			window.addEventListener("portfolio:language-change", (event) => {
				applyTranslations(event.detail?.language);
			});

			window.addEventListener("storage", (event) => {
				if (
					event.key === languageStorageKey &&
					event.newValue &&
					isSupportedLanguage(event.newValue)
				) {
					applyTranslations(event.newValue, false);
				}
			});

			const initialLanguage = isSupportedLanguage(document.documentElement.lang)
				? document.documentElement.lang
				: "es";
			applyTranslations(initialLanguage, false);
